definition{
  name: validLifecycleTransition,
  parameters: [src : MachineLifecycleState, dst : MachineLifecycleState],
  result: Bool,
  body: (match src
    | MachineLifecycleState.unverifiedDraft =>
        (match dst
          | MachineLifecycleState.dagAdmitted => true
          | MachineLifecycleState.invariantRefused => true
          | _ => false)
    | MachineLifecycleState.dagAdmitted =>
        (match dst
          | MachineLifecycleState.coherentSuperposition => true
          | _ => false)
    | MachineLifecycleState.coherentSuperposition =>
        (match dst
          | MachineLifecycleState.coherentSuperposition => true
          | MachineLifecycleState.projectiveCollapsed => true
          | MachineLifecycleState.rolledBackRebranched => true
          | _ => false)
    | MachineLifecycleState.projectiveCollapsed =>
        (match dst
          | MachineLifecycleState.rolledBackRebranched => true
          | MachineLifecycleState.invariantRefused => true
          | _ => false)
    | MachineLifecycleState.rolledBackRebranched =>
        (match dst
          | MachineLifecycleState.coherentSuperposition => true
          | MachineLifecycleState.projectiveCollapsed => true
          | _ => false)
    | MachineLifecycleState.invariantRefused =>
        (match dst
          | MachineLifecycleState.unverifiedDraft => true
          | _ => false)),
  axioms: []
}
